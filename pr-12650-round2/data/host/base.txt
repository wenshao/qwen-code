### shellcheck.run

      git ls-files | grep -v '^integration-tests/terminal-bench/' | grep -E '^([^.]+|.*\.(sh|zsh|bash))' | xargs file --mime-type         | grep "text/x-shellscript" | awk '{ print substr($1, 1, length($1)-1) }'         | xargs shellcheck           --check-sourced           --enable=all           --exclude=SC2002,SC2129,SC2310           --severity=style           --format=gcc           --color=never | sed -e 's/note:/warning:/g' -e 's/style:/warning:/g'
